Из чего собираем

Бесплатный стек

Карта и данные берутся из открытых проектов OpenStreetMap, Википедии и Wikidata. Тарифов у них нет, взамен они просят атрибуцию и бережную нагрузку. Генерация и озвучка уже работают в приложении.

    src/shared/api/public-client.ts
    import { API_TIMEOUT_MS, APP_VERSION, SUPPORT_EMAIL } from '@app/shared/config';
import type { ApiError } from './api-client';

const APP_USER_AGENT = `OhMyGuide/${APP_VERSION} (${SUPPORT_EMAIL})`;

type PublicRequest = {
  url: string;
  headers?: Record<string, string>;
  timeout?: number;
};

export const publicGet = async <T>({
  url,
  headers = {},
  timeout = API_TIMEOUT_MS,
}: PublicRequest): Promise<T> => {
  const controller = new AbortController();
  const timeoutId = setTimeout(() => controller.abort(), timeout);

  try {
    const response = await fetch(url, {
      headers: { 'User-Agent': APP_USER_AGENT, ...headers },
      signal: controller.signal,
    });

    if (!response.ok) {
      throw { message: `Ошибка сервера: ${response.status}`, status: response.status } satisfies ApiError;
    }

    return (await response.json()) as T;
  } finally {
    clearTimeout(timeoutId);
  }
};
    src/shared/api/wikidata-sights-api.ts
    import type { GeoPoint } from '@app/shared/lib/geo';
import { publicGet } from './public-client';

const WDQS_URL = 'https://query.wikidata.org/sparql';
const GRID_STEP = 0.02;

// Одинаковый центр у соседей — ответ приходит из кэша WDQS
const snapToGrid = (value: number): string => (Math.round(value / GRID_STEP) * GRID_STEP).toFixed(2);

const buildSightsQuery = ({ lat, lon }: GeoPoint): string => `
SELECT ?item ?itemLabel ?itemDescription ?lat ?lon ?title ?sitelinks ?image WHERE {
  SERVICE wikibase:around {
    ?item wdt:P625 ?coord .
    bd:serviceParam wikibase:center "Point(${snapToGrid(lon)} ${snapToGrid(lat)})"^^geo:wktLiteral .
    bd:serviceParam wikibase:radius "6" .
  }
  ?item wikibase:sitelinks ?sitelinks .
  FILTER(?sitelinks >= 8)
  ?article schema:about ?item ; schema:isPartOf <https://ru.wikipedia.org/> ; schema:name ?title .
  FILTER EXISTS { ?item wdt:P31/wdt:P279* ?class .
    VALUES ?class { wd:Q811979 wd:Q33506 wd:Q22698 wd:Q4989906 wd:Q174782 wd:Q570116 wd:Q839954 } }
  MINUS { ?item wdt:P31 ?type . VALUES ?type { wd:Q928830 wd:Q79007 } }
  OPTIONAL { ?item wdt:P18 ?image }
  BIND(geof:latitude(?coord) AS ?lat)
  BIND(geof:longitude(?coord) AS ?lon)
  SERVICE wikibase:label { bd:serviceParam wikibase:language "ru". }
}
ORDER BY DESC(?sitelinks)
LIMIT 60`;

export const fetchSightsApi = (position: GeoPoint): Promise<SparqlResponse> =>
  publicGet<SparqlResponse>({
    url: `${WDQS_URL}?format=json&query=${encodeURIComponent(buildSightsQuery(position))}`,
    timeout: 45_000,
  });

    src/shared/api/walking-route-api.ts
    import type { GeoPoint } from '@app/shared/lib/geo';
import { publicGet } from './public-client';

const OSRM_FOOT_URL = 'https://routing.openstreetmap.de/routed-foot/route/v1/foot';

export type WalkingRoute = {
  line: GeoJSON.LineString;
  distanceM: number;
  durationS: number;
};

type OsrmResponse = {
  routes?: Array<{ geometry: GeoJSON.LineString; distance: number; duration: number }>;
};

export const fetchWalkingRouteApi = async (from: GeoPoint, to: GeoPoint): Promise<WalkingRoute> => {
  const path = `${from.lon},${from.lat};${to.lon},${to.lat}`;
  const data = await publicGet<OsrmResponse>({
    url: `${OSRM_FOOT_URL}/${path}?overview=full&geometries=geojson`,
  });
  const route = data.routes?.[0];

  if (!route) {
    throw { message: 'Не удалось построить пешеходный маршрут', code: 'NO_ROUTE' };
  }

  return { line: route.geometry, distanceM: route.distance, durationS: route.duration };
};

    Что может пойти не так

    Риски и как их закрываем

    • Средний

      Нативная карта на New Architecture

      MapLibre RN 11 заявляет поддержку Expo 54 и RN 0.80+, проверяем на реальной сборке в первый день. Запасной путь — WebView с MapLibre GL JS: те же тайлы, те же данные, как в демо выше.

    • Средний

      Публичные сервисы работают на доверии

      Wikidata, OSRM и Overpass просят умеренную нагрузку и могут ограничить клиента. Закрываем кэшем по ячейке, User-Agent приложения и резервным ORS. При росте аудитории ставим свой прокси с кэшем на бесплатном тарифе Cloudflare Workers или поднимаем собственный OSRM.

    • Средний

      Wikidata медленно отвечает на первый запрос

      Новый район считается 8–20 секунд. Пока ждём, показываем ближайшие статьи Википедии (меньше секунды), затем подменяем список. Повторные запросы приходят из кэша.

    • Средний

      Модель может выдумать факты

      В запрос идут справка из Википедии, адрес и координаты, а в инструкции — просьба не упоминать то, в чём модель не уверена.

    • Низкий

      В выдаче встречаются улицы и станции метро

      В Wikidata отсекаем их по типу объекта, в GeoSearch — по названию и описанию.

    • Низкий

      Приватность координат

      Только разрешение «при использовании». Координаты перед запросами к спискам мест округляются, своего сервера, который бы их хранил, нет.

    • Низкий

      Расход батареи при ведении

      watchPositionAsync с шагом 10 м включается только на экране маршрута и выключается при уходе с него.

    • Низкий

      Обязательная атрибуция

      Подпись © OpenStreetMap и OpenFreeMap на карте, ссылка на статью Википедии (CC BY-SA) в карточке места, OSRM в маршруте.
